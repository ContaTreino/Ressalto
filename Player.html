s.close() return ip except: return "127.0.0.1" def make_png(color=(20, 20, 80)): def chunk(name, data): c = struct.pack('>I', len(data)) + name + data return c + struct.pack('>I', zlib.crc32(name + data) & 0xffffffff) w, h = 48, 48 raw = b''.join(b'\x00' + bytes(color) * w for _ in range(h)) ihdr = struct.pack('>IIBBBBB', w, h, 8, 2, 0, 0, 0) return (b'\x89PNG\r\n\x1a\n' + chunk(b'IHDR', ihdr) +
